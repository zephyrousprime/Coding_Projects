Data visualisation library for binding data to the DOM, SVG, Canvas, and scales.

- **Docs:** https://d3js.org/getting-started
- **Download:** https://github.com/d3/d3/releases

> **Note:** D3 does not provide finished charts. It supplies low-level building blocks, so keep the data join, scales, axes, and rendering code together for each visualisation.

### Installation

**Local (UMD)**

```html
<script src="path/to/d3.v7.min.js"></script>
```

**CDN (ES module)**

```html
<script type="module">
  import * as d3 from "https://cdn.jsdelivr.net/npm/d3@7/+esm";
</script>
```

**Package manager**

```bash
npm install d3
```

### Examples

#### Basic bar chart

Binds values to SVG `rect` elements and uses scales to map data values to pixels.

```html
<svg id="chart" width="420" height="220"></svg>
```

```js
const data = [12, 28, 19, 35, 24];
const width = 420;
const height = 220;
const margin = { top: 20, right: 20, bottom: 30, left: 40 };

const x = d3.scaleBand()
  .domain(d3.range(data.length))
  .range([margin.left, width - margin.right])
  .padding(0.15);

const y = d3.scaleLinear()
  .domain([0, d3.max(data)])
  .nice()
  .range([height - margin.bottom, margin.top]);

d3.select("#chart")
  .selectAll("rect")
  .data(data)
  .join("rect")
  .attr("x", (_, index) => x(index))
  .attr("y", value => y(value))
  .attr("width", x.bandwidth())
  .attr("height", value => y(0) - y(value))
  .attr("fill", "#3498db");
```

#### Load and parse a CSV file

Loads CSV data asynchronously and converts a numeric column before rendering.

```js
const rows = await d3.csv("data/sales.csv", d => ({
  month: d.month,
  sales: Number(d.sales),
}));

console.log(d3.max(rows, d => d.sales));
```

#### Update a chart with new data

Uses `join()` so new, existing, and removed bars stay in sync with an array.

```js
function updateBars(values) {
  x.domain(d3.range(values.length));
  y.domain([0, d3.max(values)]).nice();

  d3.select("#chart")
    .selectAll("rect")
    .data(values)
    .join(
      enter => enter.append("rect").attr("fill", "#3498db"),
      update => update,
      exit => exit.remove(),
    )
    .transition()
    .duration(300)
    .attr("x", (_, index) => x(index))
    .attr("y", value => y(value))
    .attr("width", x.bandwidth())
    .attr("height", value => y(0) - y(value));
}
```

#### Add axes to an SVG chart

Appends labelled x and y axes after defining the scales.

```js
const svg = d3.select("#chart");

svg.append("g")
  .attr("transform", `translate(0,${height - margin.bottom})`)
  .call(d3.axisBottom(x).tickFormat(index => `Item ${index + 1}`));

svg.append("g")
  .attr("transform", `translate(${margin.left},0)`)
  .call(d3.axisLeft(y));
```
